import { createRequire } from 'node:module';
import { spawn, spawnSync } from 'node:child_process';
const repo = process.cwd();
const require = createRequire(repo + '/package.json');
const {Client} = require('pg');
const database = process.argv[3];
const mode = process.argv[2];
if (!/^solvi_knwui060_\d+_test$/.test(database)) throw new Error('Test database name required');
const admin = new URL(process.env.DATABASE_ADMIN_URL);
if (!['localhost','127.0.0.1','::1'].includes(admin.hostname)) throw new Error('Loopback database required');
const env = { ...process.env, OTEL_EXPORTER_OTLP_ENDPOINT:'', NODE_ENV:'development' };
for (const key of ['DATABASE_URL','DATABASE_ADMIN_URL']) {
  const url = new URL(env[key]);
  if (url.hostname !== admin.hostname) throw new Error('Database hosts differ');
  url.pathname = '/' + database;
  env[key] = url.href;
}
function run(path, args, quiet=false) {
  const result = spawnSync(process.execPath, [path,...args], {env,stdio:quiet?['ignore','ignore','inherit']:'inherit'});
  if (result.status!==0) throw new Error('Isolated setup failed in ' + path + ' (' + result.status + ')');
}
if (mode==='prepare') {
  const db = new Client({connectionString:admin.href});
  await db.connect();
  try { await db.query('CREATE DATABASE "' + database + '"'); } finally { await db.end(); }
  run('tools/migrate.mjs',['up']);
  run('node_modules/tsx/dist/cli.mjs',['tools/seed.mjs'],true);
  console.log('Migrated and seeded an isolated synthetic database');
}
if (mode==='api') {
  const child=spawn(process.execPath,['node_modules/tsx/dist/cli.mjs','services/api/src/main.ts'],
    {env:{...env,API_PORT:'3261',WEB_BASE_URL:'http://127.0.0.1:3260'},stdio:'inherit'});
  for(const sig of ['SIGINT','SIGTERM']) process.on(sig,()=>child.kill(sig));
  child.on('close',code=>{process.exitCode=code??1});
}
