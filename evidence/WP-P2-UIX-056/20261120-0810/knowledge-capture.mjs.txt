const normalizeBody = (value) => value.replace(/\r\n/g, "\n");

const search = (term, manage = false) =>

  main(page).locator('a[href^="/knowledge/' + id + '"]');
const urlQuery = (page, term) =>
  page.waitForURL(
    (url) =>
      url.pathname === "/knowledge" && url.searchParams.get("q") === term,
  );

async function pageFor(role, width, colorScheme = "light") {
  const ctx = await browser.newContext({
    viewport: { width, height: width === 375 ? 812 : 900 },
    locale: "ja-JP",
    colorScheme,
    isMobile: width === 375,
    hasTouch: width === 375,
  });
  await ctx.addCookies([{ ...accounts[role], url: web }]);
  const page = await ctx.newPage();
  page.setDefaultTimeout(12000);
  return page;
}
async function enter(page, path, status = 200) {
  const response = await page.goto(web + path);
  assert.equal(response.status(), status, path);
  assert.equal(new URL(page.url()).pathname, new URL(path, web).pathname);
  await page.locator("main h1").waitFor();
}
async function use(locator, width) {
  if (width === 375) await locator.tap();
  else {
    await locator.focus();
    await locator.press("Enter");
  }
}
async function scan(page, width, theme, name) {
  assert.equal(await page.locator("html").getAttribute("data-theme"), theme);
  const overflow = await page.evaluate(
    () => document.documentElement.scrollWidth - innerWidth,
  );
  const result = await new AxeBuilder({ page }).withTags(tags).analyze();
  const violations = result.violations.map(({ id, impact, nodes }) => ({
    id,
    impact,
    nodes: nodes.map(({ target, failureSummary }) => ({
      target,
      failureSummary,
    })),
  }));
  const incomplete = result.incomplete.map(({ id, nodes }) => ({
    id,
    targets: nodes.map(({ target }) => target),
  }));
  summary.scans.push({ name, theme, width, overflow, violations, incomplete });
  await page.screenshot({
    path: join(out, "screens", [theme, width, name].join("-") + ".png"),
    animations: "disabled",
    fullPage: true,
  });
  console.log(
    "AXE",
    theme,
    width,
    name,
    overflow,
    violations.map((v) => v.id).join(",") || "pass",
  );
}
async function inspect(page, role, width, name) {
  await scan(page, width, "light", name);
  const dark = await pageFor(role, width, "dark");
  try {
    await enter(dark, page.url().slice(web.length));
    await scan(dark, width, "dark", name);
  } finally {
    await dark.context().close();
  }
}
async function read(id, role, opts = {}) {
  const { manage = false, status = 200 } = opts;
  const { article } = await api(
    "/knowledge/" + id + (manage ? "?manage=1" : ""),
    accounts[role],
    undefined,
    status,
  );
  return article;
}
async function expectArticle(page, fixture, audience, version) {
  await heading(page, fixture.title).waitFor();
  const info = main(page).getByRole("complementary", {
    name: "この記事の情報",
  });
  await info.getByText(audience, { exact: true }).waitFor();
  await info.getByText("版 " + version, { exact: true }).waitFor();
  assert.equal(
    normalizeBody(await main(page).locator(".knowledge-prose").innerText()),
    audience === "組織内" ? fixture.organizationBody : fixture.staffBody,
  );
}
async function expectZero(page, term, id) {
  await heading(page, "条件に合う記事はありません").waitFor();
  await page.getByRole("status").getByText("0件", { exact: true }).waitFor();
  assert.equal(await page.locator("#knowledge-q").inputValue(), term);
  assert.equal(await articleLink(page, id).count(), 0);
  assert.equal(await main(page).getByRole("alert").count(), 0);
  assert.match(
    await link(page, "障害を報告する").getAttribute("href"),
    /^\/tickets\/new\?kind=incident/,
  );
}
async function openFromSearch(page, width, term, id, { manage = false } = {}) {
  await enter(page, search(term, manage));
  await use(articleLink(page, id), width);
  await page.waitForURL((url) => url.pathname === "/knowledge/" + id);
}
