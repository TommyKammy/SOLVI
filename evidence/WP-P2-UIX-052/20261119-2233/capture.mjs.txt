async function contextFor(cookie, width, scheme = 'light') {
  const context = await browser.newContext({
    viewport: { width, height: width >= 1000 ? 900 : 812 },
    locale: 'ja-JP', colorScheme: scheme, isMobile: width <= 375, hasTouch: width <= 375,
  });
  if (cookie) await context.addCookies([{ ...cookie, url: web }]);
  return context;
}
async function enter(page, path) {
  const response = await page.goto(web + path);
  assert.equal(response.status(), 200, path);
  const actualPath = new URL(page.url()).pathname;
  assert.ok(!actualPath.includes('/login'), path);
  if (path !== '/') assert.equal(actualPath, new URL(path,web).pathname);
  await title(page).waitFor();
}
async function activate(target, width) {
  if (width <= 375) await target.tap();
  else {
    await target.focus();
    await target.press('Enter');
  }
}

async function scan(page, entry, width, theme, picture) {
  const {id, path} = entry;
  await enter(page, path);
  if (id === 'assistance') {
    await page.locator('#ticket-assistance').evaluate(el => { el.open = true; });
  }
  const css = await page.evaluate(() => ({
    overflow: document.documentElement.scrollWidth - innerWidth,
    culprit: Array.from(document.querySelectorAll('main *')).filter(e => {
      const r = e.getBoundingClientRect();
      return r.width > 0 && r.right > innerWidth + 1 &&
        !e.closest('.table-scroll, .directory-table-scroll, .fs-ticket-property-table');
    }).slice(0, 5).map(e => e.tagName + '.' + (typeof e.className === 'string' ? e.className : '')),
    title: document.querySelector('main h1')?.textContent?.trim(),
    h1: getComputedStyle(document.querySelector('main h1')).fontSize,
    body: getComputedStyle(document.body).fontSize,
    theme: document.documentElement.dataset.theme,
  }));
  assert.equal(css.theme, theme);
  const axe = await new AxeBuilder({ page }).withTags(tags).analyze();
  const issues = (xs) => xs.map(({id, nodes}) => ({
    id, nodes: nodes.map(({target, failureSummary}) => ({target, failureSummary})),
  }));
  summary.pages.push({id, width, theme, ...css, violations: issues(axe.violations),
    incomplete: issues(axe.incomplete)});
  if (picture) {
    await page.evaluate(()=>scrollTo(0, 0));
    await page.screenshot({path: join(out, 'screens', `live-${theme}-${width}-${id}.png`),
      animations:'disabled', fullPage: false});
  }
  console.log('PAGE', theme, width, id, css.h1, css.overflow, axe.violations.map(x=>x.id).join(',') || 'pass');
}
async function geometry(page, isMock) {
  const selectors = isMock
    ? {h1:'main #page-title', toolbar:'.toolbar', rail:'.sidebar', content:'main.page'}
    : {h1:'main h1', toolbar:'.fs-topbar', rail:'.fs-rail', content:'.fs-content'};
  await page.locator(selectors.h1).waitFor();
  const result = await page.evaluate(({h1, toolbar, rail, content})=>{
    const h = document.querySelector(h1), t = document.querySelector(toolbar);
    const sidebar = document.querySelector(rail), c = document.querySelector(content);
    const text = getComputedStyle(h), body = getComputedStyle(document.body);
    const r = h.getBoundingClientRect();
    return { label: h.textContent.trim(), titleSize:text.fontSize, titleWeight:text.fontWeight,
      titleLeading:text.lineHeight, top:r.top, left:r.left, right:innerWidth-r.right,
      font:body.fontFamily, body:body.fontSize, leading:body.lineHeight,
      bar:t.getBoundingClientRect().height, rail:sidebar.getBoundingClientRect().width,
      gutter:getComputedStyle(c).paddingInlineStart };
  },selectors);
  const cdp = await page.context().newCDPSession(page);
  await cdp.send('DOM.enable');
  await cdp.send('CSS.enable');
  const {root} = await cdp.send('DOM.getDocument');
  const {nodeId} = await cdp.send('DOM.querySelector',{nodeId:root.nodeId,selector:selectors.h1});
  const fonts = await cdp.send('CSS.getPlatformFontsForNode',{nodeId});
  await cdp.detach();
  result.drawnFonts = fonts.fonts.map(({postScriptName})=>postScriptName);
  return result;
}
async function compareOverview(staff, width) {
  const liveContext = await contextFor(staff, width);
  const referenceContext = await contextFor(null, width);
  try {
    const live = await liveContext.newPage();
    const ref = await referenceContext.newPage();
    await enter(live, '/ops/overview');
    await ref.goto(new URL('#home',mock).href);
    const reference = await geometry(ref, true);
    const actual = await geometry(live, false);
    const measured = ['titleSize','titleWeight','titleLeading','top','left','font','body','leading','bar','gutter','drawnFonts'];
    if (width === 1440) {
      measured.push('rail');
      const filter = await live.locator('.home-group-filter').evaluate(node => ({
        input: node.querySelector('select').getBoundingClientRect().bottom,
        submit: node.querySelector('button').getBoundingClientRect().bottom,
      }));
      assert.equal(filter.input, filter.submit, 'group filter action alignment');
      actual.groupFilterAligned = true;
    }
    for (const k of measured) assert.deepEqual(actual[k], reference[k], `${width} ${k}`);
    assert.equal(await live.locator('.home-layout').evaluate(e=>getComputedStyle(e).gridTemplateColumns.split(' ').length),
      width === 1440 ? 2 : 1);
    summary.geometry.push({width, reference, actual, measured});
    await ref.screenshot({path:join(out,'screens',`mock-light-${width}-overview.png`), animations:'disabled'});
  } finally { await liveContext.close(); await referenceContext.close(); }
}
// Pair static reference pages with the closest real workflow screen. The app

async function compareReferencePages(width) {
  const references = [
    ['inbox','inbox'], ['tickets','tickets'], ['portal','portal'], ['portal-list','my-tickets'],
    ['portal-report','report'], ['catalog','catalog'], ['catalog-manage','catalog-admin'],
    ['knowledge','knowledge'], ['assets','assets'], ['asset-import','asset-import-new'],
    ['admin','admin'], ['reports','reports'], ['assistance','ticket/INC-1048?focus=ai'],
  ];
  const context = await contextFor(null, width);
  try {
    const page = await context.newPage();
    for (const [id, hash] of references) {
      await page.goto(new URL('#'+hash,mock).href);
      const heading = page.locator('main h1:visible').first();
      await heading.waitFor();
      const actual = summary.pages.find(p=>p.id === id && p.theme === 'light' && p.width === width);
      const expectedSize = await heading.evaluate(h=>getComputedStyle(h).fontSize);
      assert.equal(actual.h1, expectedSize, 'reference size '+id+' '+width);
      summary.references.push({id, width, hash, headingSize: expectedSize, liveHeadingSize: actual.h1});
      if (['inbox','portal'].includes(id)) {
        await page.evaluate(()=>scrollTo(0,0));
        await page.screenshot({path:join(out,'screens',`mock-light-${width}-${id}.png`),animations:'disabled'});
      }
    }
  } finally { await context.close(); }
}

  try {

      await page.getByRole('navigation',{name:'業務メニュー'}).getByRole('link',{name:'運営状況',exact:true}).waitFor();

      await page.waitForFunction(() => document.activeElement?.getAttribute('aria-label')==='メニューを開く');

    } else {
      await page.keyboard.press('Control+k');
    }
    const search = page.locator('#fs-global-search-input');
    await search.waitFor();
    await page.waitForFunction(() => document.activeElement?.id==='fs-global-search-input');
    await search.fill(fixtures.incident.number);
    await search.press('Enter');
    await page.waitForURL(u=>u.pathname==='/ops' && u.searchParams.get('q')===fixtures.incident.number);

    await activate(entry, width);
