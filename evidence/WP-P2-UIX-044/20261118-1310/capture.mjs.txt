async function login(email) {
  const r = await fetch(`${API}/auth/login`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ email, password: PASSWORD, organizationId: ORG }),
  });
  if (!r.ok) throw new Error(`login ${email}: ${r.status}`);
  const pair = (r.headers.get('set-cookie') ?? '').split(';')[0];
  const i = pair.indexOf('=');
  return { name: pair.slice(0, i), value: pair.slice(i + 1) };
}

const cookieHeader = (c) => `${c.name}=${c.value}`;

async function json(path, cookie, options = {}) {
  const r = await fetch(API + path, {
    ...options,
    headers: {
      'content-type': 'application/json',
      ...(cookie ? { cookie: cookieHeader(cookie) } : {}),
      ...(options.headers ?? {}),
    },
  });
  const body = await r.json().catch(() => ({}));
  return { status: r.status, body };
}

const PAGES = [

const THEMES = ['light', 'dark'];

const summary = { scans: [], checks: {} };

for (const theme of THEMES) {
  const browser = await chromium.launch();
  for (const width of [1440, 375]) {
    for (const p of PAGES) {
      const ctx = await browser.newContext({
        viewport: { width, height: width === 375 ? 812 : 900 },
        locale: 'ja-JP',
        colorScheme: theme,
      });
      if (p.who) {
        await ctx.addCookies([
          { ...p.who, domain: '127.0.0.1', path: '/', httpOnly: true, secure: false, sameSite: 'Lax' },
        ]);
      }
      const page = await ctx.newPage();
      const consoleErrors = [];
      page.on('console', (m) => m.type() === 'error' && consoleErrors.push(m.text()));
      page.on('pageerror', (e) => consoleErrors.push(String(e)));
      const resp = await page.goto(WEB + p.path, { waitUntil: 'networkidle' });
      const finalPath = new URL(page.url()).pathname;
      const res = await new AxeBuilder({ page }).withTags(TAGS).analyze();
      const overflow = await page.evaluate(
        () => document.documentElement.scrollWidth - document.documentElement.clientWidth,
      );
      const name = `${theme}-${width}-${p.slug}`;
      writeFileSync(
        join(OUT, 'axe', `${name}.json`),
        JSON.stringify({ theme, width, url: p.path, finalPath, violations: res.violations }, null, 2),
      );
      await page.screenshot({ path: join(OUT, 'screens', `${name}.png`), fullPage: true });
      summary.scans.push({
        theme,
        width,
        page: p.slug,
        status: resp?.status(),
        finalPath,
        themeAttribute: await page.evaluate(() => document.documentElement.dataset.theme),
        violations: res.violations.map((v) => ({ id: v.id, impact: v.impact, nodes: v.nodes.length })),
        horizontalOverflowPx: overflow,
        consoleErrors,
      });
      await ctx.close();
    }
  }
  await browser.close();
}
