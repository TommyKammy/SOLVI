async function login(email) {
  const response = await fetch(`${API}/auth/login`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ email, password: PASSWORD, organizationId: ORG }),
  });
  if (!response.ok) throw new Error(`login ${email}: ${response.status}`);
  const pair = (response.headers.get('set-cookie') ?? '').split(';')[0];
  const index = pair.indexOf('=');
  return { name: pair.slice(0, index), value: pair.slice(index + 1) };
}

async function scan({ browser, base, theme, width, slug, path, dir, axe = true }) {
  const context = await browser.newContext({
    viewport: { width, height: width === 375 ? 812 : 900 },
    locale: 'ja-JP',
    colorScheme: theme,
    hasTouch: width === 375,
  });
  await context.addCookies([
    { ...admin, domain: '127.0.0.1', path: '/', httpOnly: true, secure: false, sameSite: 'Lax' },
  ]);
  const page = await context.newPage();
  const consoleErrors = [];
  page.on('console', (message) => {
    if (message.type() === 'error') consoleErrors.push(message.text());
  });
  page.on('pageerror', (error) => consoleErrors.push(String(error)));
  const response = await page.goto(`${base}${path}`, { waitUntil: 'networkidle' });
  const finalPath = new URL(page.url()).pathname + new URL(page.url()).search;
  const result = axe ? await new AxeBuilder({ page }).withTags(TAGS).analyze() : null;
  const layout = await page.evaluate(() => {
    const region = document.querySelector('.table-scroll');
    return {
      horizontalOverflowPx:
        document.documentElement.scrollWidth - document.documentElement.clientWidth,
      table: region
        ? {
            clientWidth: region.clientWidth,
            scrollWidth: region.scrollWidth,
            scrollsLocally: region.scrollWidth > region.clientWidth,
          }
        : null,
      pressed: [...document.querySelectorAll('[aria-pressed="true"]')].map((element) =>
        element.textContent?.replace(/\s+/g, ' ').trim(),
      ),
    };
  });
  const name = `${theme}-${width}-${slug}`;
  if (result)
    writeFileSync(
      join(OUT, 'axe', `${name}.json`),
      JSON.stringify(
        { theme, width, url: path, base, finalPath, violations: result.violations },
        null,
        2,
      ),
    );
  await page.screenshot({ path: join(OUT, dir, `${name}.png`), fullPage: true });
  const scanResult = {
    theme,
    width,
    slug,
    base,
    path,
    status: response?.status(),
    finalPath,
    themeAttribute: await page.evaluate(() => document.documentElement.dataset.theme),
    violations: result
      ? result.violations.map((violation) => ({
          id: violation.id,
          impact: violation.impact,
          nodes: violation.nodes.length,
        }))
      : [],
    horizontalOverflowPx: layout.horizontalOverflowPx,
    table: layout.table,
    pressed: layout.pressed,
    consoleErrors,
  };
  summary.scans.push(scanResult);
  await context.close();
  return scanResult;
}

// TL-12: real login -> inbox -> list filter -> ticket detail -> reply send.
{
  const browser = await chromium.launch();
  const context = await browser.newContext({
    viewport: { width: 1440, height: 900 },
    locale: 'ja-JP',
    colorScheme: 'light',
  });
  const page = await context.newPage();
  const steps = [];
  const started = Date.now();
  const step = async (name) => {
    steps.push({
      name,
      url: page.url(),
      path: new URL(page.url()).pathname + new URL(page.url()).search,
      elapsedMs: Date.now() - started,
    });
    await page.screenshot({
      path: join(OUT, 'tl12', `${String(steps.length).padStart(2, '0')}-${name}.png`),
      fullPage: true,
    });
  };
  await page.goto(`${AFTER}/login`, { waitUntil: 'networkidle' });
  await step('login');
  await page.fill('#email', 'org_admin@acme.example.test');
  await page.fill('#password', PASSWORD);
  await Promise.all([
    page.waitForURL((url) => url.pathname === '/inbox', { timeout: 15000 }),
    page.click('button[type="submit"]'),
  ]);
  await page.waitForLoadState('networkidle');
  await step('after-login-inbox');
  await page.click('#solvi-main-navigation a[href="/ops"]');
  await page.waitForURL((url) => url.pathname === '/ops', { timeout: 15000 });
  await page.waitForLoadState('networkidle');
  await step('ticket-list');
  const chip = page.locator('.queue-metric', { hasText: '担当者未割当' });
  const chipState = {
    before: await chip.getAttribute('aria-pressed'),
    href: await chip.getAttribute('href'),
  };
  await chip.click();
  await page.waitForURL((url) => url.searchParams.get('assignment') === 'unassigned', {
    timeout: 15000,
  });
  await page.waitForLoadState('networkidle');
  chipState.after = await chip.getAttribute('aria-pressed');
  await step('filtered-list');
  const firstSubject = page.locator('.queue-subject').first();
  const subject = await firstSubject.innerText();
  const detailHref = await firstSubject.getAttribute('href');
  await Promise.all([
    page.waitForURL((url) => /^\/ops\/[0-9a-f-]{36}$/.test(url.pathname), { timeout: 15000 }),
    firstSubject.click(),
  ]);
  await page.waitForLoadState('networkidle');
  await step('ticket-detail');
  const beforeCommentCount = await page.locator('.comments .comment').count();
  const beforeHistoryCount = await page.locator('.ticket-history li').count();
  await page.fill(
    '#reply-body',
    'WP-P2-UIX-046 のTL-12で送信する合成の返信です。実在の人物・組織とは関係ありません。',
  );
  await page.click('button[type="submit"][name="outcome"][value="reply"]');
  await page.waitForSelector('text=返信を保存しました。', { timeout: 15000 });
  await page.waitForLoadState('networkidle');
  await step('reply-sent');
  const detail = {
    subject,
    detailHref,
    url: page.url(),
    success: await page.locator('text=返信を保存しました。').first().innerText(),
    beforeCommentCount,
    afterCommentCount: await page.locator('.comments .comment').count(),
    beforeHistoryCount,
    afterHistoryCount: await page.locator('.ticket-history li').count(),
  };
  summary.checks.tl12 = { steps, chipState, detail };
  await context.close();
  await browser.close();
}

    (total, scan) =>

  ),

    (total, scan) => total + scan.violations.reduce((all, violation) => all + violation.nodes, 0),

  afterTableLocalScroll: summary.scans

      slug: scan.slug,
      theme: scan.theme,
      clientWidth: scan.table.clientWidth,
      scrollWidth: scan.table.scrollWidth,
      scrollsLocally: scan.table.scrollsLocally,

  redirectedToLogin: summary.scans

    .map((scan) => `${scan.theme}-${scan.width}-${scan.slug}`),
  consoleErrorScans: summary.scans
    .filter((scan) => scan.consoleErrors.length > 0)

  themeAttributeMismatches: summary.scans
    .filter((scan) => scan.themeAttribute !== scan.theme)
    .map((scan) => `${scan.theme}-${scan.width}-${scan.slug}:${scan.themeAttribute}`),
};
