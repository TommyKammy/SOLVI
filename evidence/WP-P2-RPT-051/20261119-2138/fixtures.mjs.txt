import assert from 'node:assert/strict';
import { randomUUID } from 'node:crypto';
import { api, login, require } from './helpers.mjs';
const { Client } = require('pg');

export async function testDatabase(database) {
  assert.match(database, /^solvi_rpt051_\d+_test$/);
  const uri = new URL(process.env.DATABASE_ADMIN_URL);
  assert.ok(['localhost', '127.0.0.1', '::1'].includes(uri.hostname));
  uri.pathname = '/' + database;
  const client = new Client({ connectionString: uri.href });
  await client.connect();
  return client;
}

export async function accountsFor(org) {
  const roles = ['org_admin', 'auditor', 'agent', 'approver', 'requester'];
  return Object.fromEntries(await Promise.all(roles.map(async (role) => [role, await login(role, org)])));
}

export async function ticketFixture(db, accounts, suffix) {
  const session = await api('/auth/me', accounts.agent);
  const group = await api('/groups', accounts.org_admin, {
    code: 'rpt-' + suffix,
    name: '検証用サポート ' + suffix,
  });
  const tickets = {};
  for (const [key, name, state, elapsed] of [
    ['breached', '受付端末のネットワーク障害', 'new', 2000000],
    ['fresh', '貸出機器の設定相談', 'new', 0],
    ['paused', '利用者の回答を待つ接続障害', 'pending', 2000000],
    ['resolved', '印刷エラーの復旧完了', 'resolved', 0],
  ]) {
    const subject = name + '（' + suffix + '）';
    const ticket = await api('/tickets', accounts.requester, {
      kind: 'incident',
      subject,
      body: '隔離環境でのレポート確認用合成チケット',
      impact: 'low',
      urgency: 'low',
    });
    const changed = await db.query(
      `UPDATE ticket SET assignee_group_id=$1, state=$2, sla_elapsed_seconds=$3,
        sla_clock_started_at=CASE WHEN $2='new' THEN clock_timestamp() ELSE NULL END,
        resolved_at=CASE WHEN $2='resolved' THEN clock_timestamp() ELSE NULL END
      WHERE id=$4 AND organization_id=$5`,
      [group.id, state, elapsed, ticket.id, session.organizationId],
    );
    assert.equal(changed.rowCount, 1);
    tickets[key] = { id: ticket.id, number: ticket.number, subject };
  }
  return { group, tickets };
}

export async function roleFixture(db, accounts, suffix) {
  const requester = await api('/auth/me', accounts.requester);
  const path = '/users/' + requester.userId + '/roles';
  const body = { roleCode: 'agent', reason: 'SYNTH_PRIVATE_ROLE_' + suffix };
  await api(path, accounts.org_admin, body, 204);
  await api(path + '/revoke', accounts.org_admin, body, 204);
  // Obtain the binding target for the real records. Check their exact actor-independent subject and org.
  const { rows } = await db.query(
    `SELECT event_id, event_type, target_id FROM audit_event
        WHERE organization_id=$1 AND subject_user_id=$2
          AND event_type IN ('role.binding.created','role.binding.deleted')
          AND (after_state->>'roleCode'='agent' OR before_state->>'roleCode'='agent')
        ORDER BY occurred_at DESC LIMIT 2`,
    [requester.organizationId, requester.userId],
  );
  assert.deepEqual(rows.map(({ event_type }) => event_type), ['role.binding.deleted', 'role.binding.created']);
  assert.equal(rows[0].target_id, rows[1].target_id);
  return { target: rows[0].target_id, event: rows[0].event_id };
}

export async function auditReads(db, accounts) {
  const auditor = await api('/auth/me', accounts.auditor);
  const { rows } = await db.query(
    `SELECT after_state FROM audit_event WHERE organization_id=$1 AND actor_id=$2
       AND event_type='audit.viewed' ORDER BY occurred_at,event_id`,
    [auditor.organizationId, auditor.userId],
  );
  return rows.map(({ after_state }) => after_state);
}
