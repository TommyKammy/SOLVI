import assert from 'node:assert/strict';
import { writeFileSync } from 'node:fs';
import { login, require, web } from './helpers.mjs';

const { chromium } = require('playwright');
const { AxeBuilder } = require('@axe-core/playwright');
const account = await login('org_admin');
const browser = await chromium.launch({ channel: 'chrome' });
const rows = [];

try {
  for (const colorScheme of ['light', 'dark']) {
    const context = await browser.newContext({
      colorScheme, viewport: { width: 375, height: 812 }, hasTouch: true, isMobile: true,
    });
    await context.addCookies([{ ...account, url: web }]);
    const page = await context.newPage();
    for (const path of ['/inbox?kind=ticket&q=UIX048&page=2', '/ops?keyword=VPN&state=in_progress']) {
      const response = await page.goto(web + path);
      assert.equal(response.status(), 200);
      const result = await new AxeBuilder({ page }).withRules(['color-contrast']).analyze();
      const incomplete = result.incomplete.find((entry) => entry.id === 'color-contrast');
      const targets = incomplete?.nodes.map((node) => node.target) ?? [];
      for (const target of targets) {
        assert.equal(target.length, 1);
        assert.equal(typeof target[0], 'string');
        const measured = await page.locator(target[0]).evaluate((element) => {
          const canvas = document.createElement('canvas');
          canvas.width = canvas.height = 1;
          const ctx = canvas.getContext('2d', { willReadFrequently: true });
          const parse = (color) => {
            ctx.clearRect(0, 0, 1, 1);
            ctx.fillStyle = color;
            ctx.fillRect(0, 0, 1, 1);
            const pixel = ctx.getImageData(0, 0, 1, 1).data;
            return [pixel[0], pixel[1], pixel[2], pixel[3] / 255];
          };
          const blend = (front, back) => back.map((part, i) => front[i] * front[3] + part * (1 - front[3]));
          const luma = (rgb) => rgb.reduce((sum, component, i) => {
            const unit = component / 255;
            const linear = unit <= 0.04045 ? unit / 12.92 : ((unit + 0.055) / 1.055) ** 2.4;
            return sum + linear * [0.2126, 0.7152, 0.0722][i];
          }, 0);
          const backgrounds = [];
          for (let node = element; node; node = node.parentElement) {
            const style = getComputedStyle(node);
            if (style.backgroundImage !== 'none' || style.opacity !== '1') {
              throw new Error('Non-solid background requires review');
            }
            const background = parse(style.backgroundColor);
            backgrounds.push(background);
            if (background[3] === 1) break;
          }
          if (backgrounds.at(-1)[3] !== 1) throw new Error('No opaque background');
          const background = backgrounds.reduceRight((result, color) => blend(color, result), [0, 0, 0]);
          const fg = luma(blend(parse(getComputedStyle(element).color), background));
          const bg = luma(background);
          return { ratio: Math.round((Math.max(fg, bg) + 0.05) / (Math.min(fg, bg) + 0.05) * 100) / 100 };
        });
        assert(measured.ratio >= 4.5, JSON.stringify({ colorScheme, path, target, ...measured }));
        rows.push({ colorScheme, path, target: target[0], ...measured });
      }
    }
    await context.close();
  }
} finally {
  await browser.close();
}
assert(rows.length > 0, 'expected mobile table color contrast checks');
writeFileSync(process.argv[2], JSON.stringify(rows, null, 2) + '\n');
console.log('PASS mobile contrast for ' + rows.length + ' axe-incomplete nodes; minimum ' +
  Math.min(...rows.map(({ ratio }) => ratio)) + ':1');
