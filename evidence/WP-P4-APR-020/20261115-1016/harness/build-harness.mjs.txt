const summary = { harness: true, css: ['globals.css', 'freshservice.css'], startedAt: now.toISOString(), scans: [] };
for (const [name, title, main] of pages) {
  const file = resolve(output, `${name}.html`);
  await writeFile(file, frame(title, main));
  for (const engineName of ['chromium', 'webkit']) {
    const browser = await (engineName === 'chromium'
      ? chromium.launch({ channel: 'chrome' })
      : webkit.launch());
    try {
      for (const width of [1440, 375]) {
        const context = await browser.newContext({
          viewport: { width, height: 900 },
          deviceScaleFactor: 1,
        });
        const page = await context.newPage();
        await page.goto(`file://${file}`, { waitUntil: 'load' });
        const dimensions = await page.evaluate(() => ({
          document: document.documentElement.scrollWidth,
          viewport: innerWidth,
        }));
        if (dimensions.document > dimensions.viewport)
          throw new Error(`${name} overflows at ${width}: ${dimensions.document} > ${dimensions.viewport}`);
        const result = await new AxeBuilder({ page })
          .withTags(['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'])
          .analyze();
        const violations = result.violations.map(({ id, impact, nodes, help }) => ({
          id,
          impact,
          help,
          targets: nodes.map((node) => node.target),
        }));
        const severe = violations.filter(
          (violation) => violation.impact === 'serious' || violation.impact === 'critical',
        );
        const scan = { name, engine: engineName, width, dimensions, violations };
        summary.scans.push(scan);
        await writeFile(
          resolve(output, `axe-${engineName}-${width}-${name}.json`),
          JSON.stringify(scan, null, 2) + '\n',
        );
        await page.screenshot({
          path: resolve(output, `${engineName}-${width}-${name}.png`),
          fullPage: true,
        });
        if (severe.length > 0)
          throw new Error(
            `${name} ${engineName} ${width}: serious/critical axe violations: ${severe
              .map((violation) => violation.id)
              .join(', ')}`,
          );
        await context.close();
      }
    } finally {
      await browser.close();
    }
  }
}
summary.finishedAt = new Date().toISOString();
summary.severe = summary.scans.filter((scan) =>
  scan.violations.some((v) => v.impact === 'serious' || v.impact === 'critical'),
).length;
await writeFile(resolve(output, 'axe-summary.json'), JSON.stringify(summary, null, 2) + '\n');
