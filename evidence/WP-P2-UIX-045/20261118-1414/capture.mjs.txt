async function login(email) {
  const response = await fetch(`${API}/auth/login`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ email, password: PASSWORD, organizationId: ORG }),
  });
  if (!response.ok) throw new Error(`login ${email}: ${response.status}`);
  const pair = (response.headers.get('set-cookie') ?? '').split(';')[0];
  const index = pair.indexOf('=');
  return { name: pair.slice(0, index), value: pair.slice(index + 1) };
}

async function scan({
  browser,
  base,
  theme,
  width,
  slug,
  who,
  dir = 'screens',
  axe = true,
}) {
  const context = await browser.newContext({
    viewport: { width, height: width === 375 ? 812 : 900 },
    locale: 'ja-JP',
    colorScheme: theme,
    hasTouch: width === 375,
  });
  if (who)
    await context.addCookies([
      { ...who, domain: '127.0.0.1', path: '/', httpOnly: true, secure: false, sameSite: 'Lax' },
    ]);
  const page = await context.newPage();
  const consoleErrors = [];
  page.on('console', (message) => {
    if (message.type() === 'error') consoleErrors.push(message.text());
  });
  page.on('pageerror', (error) => consoleErrors.push(String(error)));
  const response = await page.goto(`${base}/ops/overview`, { waitUntil: 'networkidle' });
  const finalPath = new URL(page.url()).pathname;
  const result = axe ? await new AxeBuilder({ page }).withTags(TAGS).analyze() : null;
  const overflow = await page.evaluate(
    () => document.documentElement.scrollWidth - document.documentElement.clientWidth,
  );
  const name = `${theme}-${width}-${slug}`;
  if (result)
    writeFileSync(
      join(OUT, 'axe', `${name}.json`),
      JSON.stringify(
        { theme, width, url: '/ops/overview', base, finalPath, violations: result.violations },
        null,
        2,
      ),
    );
  await page.screenshot({ path: join(OUT, dir, `${name}.png`), fullPage: true });
  const scanResult = {
    theme,
    width,
    slug,
    base,
    status: response?.status(),
    finalPath,
    themeAttribute: await page.evaluate(() => document.documentElement.dataset.theme),
    violations: result
      ? result.violations.map((violation) => ({
          id: violation.id,
          impact: violation.impact,
          nodes: violation.nodes.length,
        }))
      : [],
    horizontalOverflowPx: overflow,
    consoleErrors,
  };
  summary.scans.push(scanResult);
  await context.close();
  return scanResult;
}

// TL-12: real login -> home -> overdue row -> ticket detail.
{
  const browser = await chromium.launch();
  const context = await browser.newContext({
    viewport: { width: 1440, height: 900 },
    locale: 'ja-JP',
    colorScheme: 'light',
  });
  const page = await context.newPage();
  const steps = [];
  const started = Date.now();
  const step = async (name, path) => {
    steps.push({ name, url: page.url(), path, elapsedMs: Date.now() - started });
    await page.screenshot({ path: join(OUT, 'tl12', `${String(steps.length).padStart(2, '0')}-${name}.png`), fullPage: true });
  };
  await page.goto(`${AFTER}/login`, { waitUntil: 'networkidle' });
  await step('login', '/login');
  await page.fill('#email', 'org_admin@acme.example.test');
  await page.fill('#password', PASSWORD);
  await Promise.all([
    page.waitForURL((url) => url.pathname === '/inbox', { timeout: 15000 }),
    page.click('button[type="submit"]'),
  ]);
  await page.waitForLoadState('networkidle');
  await step('after-login-inbox', '/inbox');
  await page.click('#solvi-main-navigation a[href="/ops/overview"]');
  await page.waitForURL((url) => url.pathname === '/ops/overview', { timeout: 15000 });
  await page.waitForLoadState('networkidle');
  await step('home', '/ops/overview');
  const overdue = page.locator('a.home-work-row[data-due="overdue"]').first();
  const overdueRow = {
    count: await page.locator('a.home-work-row[data-due="overdue"]').count(),
    href: await overdue.getAttribute('href'),
    subject: await overdue.locator('.home-work-title').innerText(),
    due: await overdue.locator('.home-work-due').innerText(),
  };
  await overdue.scrollIntoViewIfNeeded();
  await overdue.screenshot({ path: join(OUT, 'tl12', '04-overdue-row.png') });
  await Promise.all([
    page.waitForURL((url) => /^\/ops\/[0-9a-f-]{36}$/.test(url.pathname), { timeout: 15000 }),
    overdue.click(),
  ]);
  await page.waitForLoadState('networkidle');
  await step('ticket-detail', new URL(page.url()).pathname);
  const detail = {
    heading: await page.locator('h1').first().innerText(),
    url: page.url(),
    hasNumber: (await page.content()).includes(overdueRow.subject.slice(0, 12)),
  };
  summary.checks.tl12 = { steps, overdueRow, detail };
  await context.close();
  await browser.close();
}

summary.totals = {
  scans: summary.scans.length,
  seriousOrCritical: summary.scans.reduce(
    (total, scan) =>
      total +
      scan.violations
        .filter((violation) => ['serious', 'critical'].includes(violation.impact))
        .reduce((all, violation) => all + violation.nodes, 0),
    0,
  ),
  anyViolationNodes: summary.scans.reduce(
    (total, scan) => total + scan.violations.reduce((all, violation) => all + violation.nodes, 0),
    0,
  ),
  overflow375: summary.scans
    .filter((scan) => scan.width === 375 && scan.horizontalOverflowPx > 0)
    .map((scan) => `${scan.base}:${scan.theme}:${scan.slug}:${scan.horizontalOverflowPx}`),
  redirectedToLogin: summary.scans
    .filter((scan) => scan.finalPath.startsWith('/login') && scan.slug !== 'login')
    .map((scan) => `${scan.theme}-${scan.width}-${scan.slug}`),
  consoleErrorScans: summary.scans
    .filter((scan) => scan.consoleErrors.length > 0)
    .map((scan) => `${scan.theme}-${scan.width}-${scan.slug}`),
  themeAttributeMismatches: summary.scans
    .filter((scan) => scan.themeAttribute !== scan.theme)
    .map((scan) => `${scan.theme}-${scan.width}-${scan.slug}:${scan.themeAttribute}`),
};
writeFileSync(join(OUT, 'summary.json'), JSON.stringify(summary, null, 2));
