// WP-P2-UIX-045 evidence: make only /inbox fail while the rest of the API stays up.
// This exercises the home page's partial-failure rendering without stopping the API.
import http from 'node:http';

const target = new URL('http://127.0.0.1:3101');
const server = http.createServer((request, response) => {
  const path = (request.url ?? '').split('?')[0];
  if (path === '/inbox') {
    response.writeHead(503, { 'content-type': 'application/json; charset=utf-8' });
    response.end(JSON.stringify({ status: 503, title: '合成の失敗（受信箱のみ）' }));
    return;
  }
  const upstream = http.request(
    {
      hostname: target.hostname,
      port: target.port,
      path: request.url,
      method: request.method,
      headers: { ...request.headers, host: target.host },
    },
    (result) => {
      response.writeHead(result.statusCode ?? 502, result.headers);
      result.pipe(response);
    },
  );
  upstream.on('error', () => {
    response.writeHead(502, { 'content-type': 'application/json; charset=utf-8' });
    response.end(JSON.stringify({ status: 502, title: '合成の失敗' }));
  });
  request.pipe(upstream);
});

server.listen(3103, '127.0.0.1', () => {
  console.log('partial-failure proxy on 127.0.0.1:3103');
});
