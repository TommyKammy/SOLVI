async function pageFor(role, width, colorScheme = "light") {
  const context = await browser.newContext({
    viewport: { width, height: width === 375 ? 812 : 900 },
    locale: "ja-JP",
    colorScheme,
    isMobile: width === 375,
    hasTouch: width === 375,
  });
  if (role) await context.addCookies([{ ...(await login(role)), url: web }]);
  const page = await context.newPage();
  page.setDefaultTimeout(12000);
  return page;
}
async function enter(page, path) {
  const response = await page.goto(web + path);
  assert.equal(response.status(), 200, path);
  assert.equal(pathOf(page.url()), pathOf(path));
  await main(page).locator("h1").waitFor();
}
async function use(locator, width) {
  if (width === 375) await locator.tap();
  else {
    await locator.focus();
    await locator.press("Enter");
  }
}
async function scan(page, name, width, theme) {
  assert.equal(await page.locator("html").getAttribute("data-theme"), theme);
  if (['users','department','group','approval','notification-detail'].includes(name)) {
    const areas = await main(page).locator('.admin-split').evaluate(el => {
      const [list, detail] = el.children;
      const left = list.getBoundingClientRect();
      const right = detail.getBoundingClientRect();
      return { sideBySide: right.left > left.left, below: right.top >= left.bottom };
    });
    assert.equal(areas.sideBySide, width === 1440, name);
    if (width === 375) assert(areas.below, name);
  }
  const overflow = await page.evaluate(
    () => document.documentElement.scrollWidth - innerWidth,
  );
  const result = await new AxeBuilder({ page }).withTags(tags).analyze();
  const violations = result.violations.map(({ id, impact, nodes }) => ({
    id,
    impact,
    nodes: nodes.map(({ target, failureSummary }) => ({
      target,
      failureSummary,
    })),
  }));
  const incomplete = result.incomplete.map(({ id, nodes }) => ({
    id,
    targets: nodes.map(({ target }) => target),
  }));
  summary.scans.push({ name, theme, width, overflow, violations, incomplete });
  await page.evaluate(() => window.scrollTo(0, 0));
  await page.screenshot({
    path: join(out, "screens", [theme, width, name].join("-") + ".png"),
    animations: "disabled",
    fullPage: true,
  });
  console.log(
    "AXE",
    theme,
    width,
    name,
    overflow,
    violations.map((v) => v.id).join(",") || "pass",
  );
}
async function inspect(page, role, width, name, beforeDark) {
  await scan(page, name, width, "light");
  const dark = await pageFor(role, width, "dark");
  try {
    await enter(dark, page.url().slice(web.length));
    if (beforeDark) await beforeDark(dark);
    await scan(dark, name, width, "dark");
  } finally {
    await dark.context().close();
  }
}
async function appLogin(page, role, width) {
  await page.locator("#email").fill(role + "@acme.example.test");
  await page.locator("#password").fill(password);
  await use(button(page, "ログイン"), width);
}
async function loginExpiry(page, target, width, screenshot) {
  const cookies = await page.context().cookies(web);
  const cookie = cookies.find(({ name }) => name.startsWith("solvi_session"));
  assert.ok(cookie);
  await expireTestSession(db, cookie, actors.org_admin);
  await page.goto(web + target);
  await waitPath(page, "/login");
  await heading(page, "ログインの有効期限が切れました").waitFor();
  const back = new URL(page.url()).searchParams.get("returnTo");
  assert.ok(back);
  assert.equal(pathOf(back), pathOf(target));
  assert.equal(await page.locator("input[name=returnTo]").inputValue(), back);
  if (screenshot) {
    await scan(page, "expired", width, "light");
    const dark = await pageFor(null, width, "dark");
    try {
      await enter(dark, page.url().slice(web.length));
      await scan(dark, "expired", width, "dark");
    } finally {
      await dark.context().close();
    }
  }
  return back;
}

}

function findDepartment(items, code) {
  for (const item of items) {
    if (item.code === code) return item;
    const found = findDepartment(item.children ?? [], code);
    if (found) return found;
  }
  return null;
}
