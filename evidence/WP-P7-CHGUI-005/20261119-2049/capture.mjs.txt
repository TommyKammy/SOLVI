async function pageFor(role, width, colorScheme = "light") {
  const context = await browser.newContext({
    viewport: { width, height: width === 375 ? 812 : 900 },
    locale: "ja-JP",
    colorScheme,
    isMobile: width === 375,
    hasTouch: width === 375,
  });
  await context.addCookies([{ ...accounts[role], url: web }]);
  const page = await context.newPage();
  page.setDefaultTimeout(12000);
  return page;
}
async function enter(page, path) {
  const response = await page.goto(web + path);
  assert.equal(response.status(), 200, path);
  assert.equal(openPath(page.url()), new URL(path, web).pathname);
  await main(page).locator("h1").waitFor();
}
async function use(locator, width, key = "Enter") {
  if (width === 375) await locator.tap();
  else {
    await locator.focus();
    await locator.press(key);
  }
}
async function check(locator, width) {
  await use(locator, width, "Space");
  assert.ok(await locator.isChecked());
}
async function scan(page, name, width, theme) {
  assert.equal(await page.locator("html").getAttribute("data-theme"), theme);
  const overflow = await page.evaluate(
    () => document.documentElement.scrollWidth - innerWidth,
  );
  const table = region(page, "CSVの行ごとの変更と隔離理由");
  const csvTableOverflow = (await table.count())
    ? await table.evaluate(
        (element) => element.scrollWidth - element.clientWidth,
      )
    : null;
  const result = await new AxeBuilder({ page }).withTags(tags).analyze();
  const violations = result.violations.map(({ id, impact, nodes }) => ({
    id,
    impact,
    nodes: nodes.map(({ target, failureSummary }) => ({
      target,
      failureSummary,
    })),
  }));
  const incomplete = result.incomplete.map(({ id, nodes }) => ({
    id,
    targets: nodes.map(({ target }) => target),
  }));
  summary.scans.push({
    name,
    theme,
    width,
    overflow,
    csvTableOverflow,
    violations,
    incomplete,
  });
  await page.evaluate(() => window.scrollTo(0, 0));
  await page.screenshot({
    path: join(out, "screens", [theme, width, name].join("-") + ".png"),
    animations: "disabled",
    fullPage: true,
  });
  console.log(
    "AXE",
    theme,
    width,
    name,
    overflow,
    csvTableOverflow,
    violations.map((v) => v.id).join(",") || "pass",
  );
}
async function inspect(page, role, width, name, filterError = false) {
  await scan(page, name, width, "light");
  const dark = await pageFor(role, width, "dark");
  try {
    await enter(dark, page.url().slice(web.length));
    if (filterError) await use(button(dark, "不正（1）"), width);
    await scan(dark, name, width, "dark");
  } finally {
    await dark.context().close();
  }
}
async function preview(page, width, source, text) {
  await page.locator("#csv-source").fill(source);
  await page.locator("#csv-reason").fill("合成の登録内容と隔離行を確認");
  await page.locator("#csv-text").fill(text);
  await use(button(page, "列を確認する"), width);
  await heading(page, "列の対応付け").waitFor();
  assert.equal(
    await page.locator("#csv-map-externalId").inputValue(),
    "externalId",
  );
  assert.equal(
    await page.locator("#csv-map-assetTag").inputValue(),
    "assetTag",
  );
  await use(button(page, "取込プレビューを作成する"), width);
  await page.waitForURL((url) =>
    /^\/assets\/imports\/[a-f0-9-]{36}$/.test(url.pathname),
  );
  await heading(page, "CSV取込プレビュー").waitFor();
  return openPath(page.url());
}
async function matchingAssets(tag) {
  return (
    await api("/assets?q=" + encodeURIComponent(tag), accounts.agent)
  ).items.filter((asset) => asset.assetTag === tag);
}
async function assertChange(path, expectedState, assetId, ticketId) {
  const saved = await api(path, accounts.agent);
  assert.equal(saved.change.state, expectedState);
  assert.equal(saved.change.snapshot.ticket.id, ticketId);
  assert.deepEqual(
    saved.change.snapshot.assets.map(({ id, version }) => ({ id, version })),
    [{ id: assetId, version: 1 }],
  );
  return saved;
}
function jst(inMinutes) {
  return new Date(Date.now() + (9 * 60 + inMinutes) * 60000)
    .toISOString()
    .slice(0, 16);
}
