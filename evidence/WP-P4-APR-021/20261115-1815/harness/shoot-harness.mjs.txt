// WP-P4-APR-021 段階B のハーネス撮影・axe。
// 実画面ではなく、実際の globals.css / freshservice.css を読み込んだ静的HTMLを撮る。
import { mkdir, readFile, writeFile } from 'node:fs/promises';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
import { chromium, webkit } from 'playwright';
import { AxeBuilder } from '@axe-core/playwright';

const root = resolve(import.meta.dirname, '../../../..');
const output = resolve(root, process.env.WP021_OUT);
await mkdir(output, { recursive: true });

const states = [
  'catalog-admin-highrisk-unchecked',
  'catalog-admin-highrisk-checked',
  'catalog-admin-highrisk-delegated',
  'catalog-publish-highrisk-blocked',
  'request-new-highrisk',
  'request-detail-highrisk',
  'approvals-rule-highrisk-blocked',
];
const engines = {
  chromium: { launch: () => chromium.launch({ channel: 'chrome', headless: true }) },
  webkit: { launch: () => webkit.launch({ headless: true }) },
};
const summary = [];
const failures = [];

for (const [engineName, engine] of Object.entries(engines)) {
  const browser = await engine.launch();
  try {
    for (const width of [1440, 375]) {
      const context = await browser.newContext({ viewport: { width, height: 900 } });
      const page = await context.newPage();
      for (const state of states) {
        const url = pathToFileURL(resolve(output, `${state}.html`)).href;
        await page.goto(url);
        await page.waitForLoadState('load');
        const dimensions = await page.evaluate(() => ({
          document: document.documentElement.scrollWidth,
          viewport: innerWidth,
          duplicateIds: [...document.querySelectorAll('[id]')]
            .map((node) => node.id)
            .filter((id, index, ids) => ids.indexOf(id) !== index),
        }));
        const result = await new AxeBuilder({ page })
          .withTags(['wcag2a', 'wcag2aa', 'wcag21aa', 'wcag22aa'])
          .analyze();
        const violations = result.violations.map(({ id, impact, nodes }) => ({
          id,
          impact,
          targets: nodes.map((node) => node.target),
        }));
        const serious = violations.filter(
          (violation) => violation.impact === 'serious' || violation.impact === 'critical',
        );
        const name = `${engineName}-${width}-${state}`;
        await page.screenshot({ path: resolve(output, `${name}.png`), fullPage: true });
        await writeFile(resolve(output, `axe-${name}.json`), JSON.stringify(violations, null, 2));
        const entry = {
          engine: engineName,
          width,
          state,
          dimensions,
          violations,
          seriousOrCritical: serious.length,
        };
        summary.push(entry);
        if (dimensions.document > dimensions.viewport)
          failures.push(`${name}: horizontal overflow ${dimensions.document} > ${width}`);
        if (dimensions.duplicateIds.length > 0)
          failures.push(`${name}: duplicate ids ${dimensions.duplicateIds.join(',')}`);
        if (serious.length > 0)
          failures.push(
            `${name}: axe serious/critical ${serious.map((v) => v.id).join(',')}`,
          );
      }
      await context.close();
    }
  } finally {
    await browser.close();
  }
}

await writeFile(
  resolve(output, 'axe-summary.json'),
  JSON.stringify(
    {
      harness: true,
      note: '静的HTMLハーネス。実画面ではない。globals.css と freshservice.css を読み込む。',
      scans: summary.length,
      seriousOrCritical: summary.reduce((sum, entry) => sum + entry.seriousOrCritical, 0),
      horizontalOverflow: failures.filter((failure) => failure.includes('horizontal overflow')).length,
      failures,
      results: summary,
    },
    null,
    2,
  ),
);
console.log(JSON.stringify({ scans: summary.length, failures }, null, 2));
if (failures.length > 0) process.exitCode = 1;
